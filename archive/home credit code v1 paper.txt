def jaccard(s1, s2):
    union = s1 | s2
    if not union:
        return 1.0
    return len(s1 & s2) / len(union)


def structural_complexity(feature_name_list):
    """
    SC = 0.4*(F/10) + 0.4*(Nc/10) + 0.2*(L¯/20)

    F  = number of features shown (always K=10)
    Nc = features whose name contains a conditional operator
         (>, <, <=, >=, =) — nonzero only for LIME condition strings
    L¯  = mean character length of feature name strings

    Lower SC = simpler = better compliance.
    Paper Section III-C, Eq. (3). Nc/10 (not Nc/F) because F=10 always.
    """
    names = [str(n) for n in feature_name_list]
    F     = len(names)
    Nc    = sum(1 for n in names
                if any(op in n for op in ['>', '<', '<=', '>=', '=']))
    L_bar = float(np.mean([len(n) for n in names])) if names else 0.0
    return 0.4 * (F / 10) + 0.4 * (Nc / 10) + 0.2 * (L_bar / 20)


# Score-conversion functions — Table II thresholds
def score_coverage(v):
    if v >= 0.90: return 5.0
    if v >= 0.80: return 4.0
    if v >= 0.70: return 3.0
    if v >= 0.60: return 2.0
    return 1.0

def score_fidelity(v):
    if v >= 0.85: return 5.0
    if v >= 0.75: return 4.0
    if v >= 0.65: return 3.0
    if v >= 0.55: return 2.0
    return 1.5   # non-compliant but non-zero credit

def score_jaccard(v):
    if v >= 0.95: return 5.0
    if v >= 0.85: return 4.0
    if v >= 0.75: return 3.0
    if v >= 0.60: return 2.0
    return 1.0

def score_complexity(v):   # lower = better
    if v <= 0.40: return 5.0
    if v <= 0.55: return 4.0
    if v <= 0.70: return 3.0
    if v <= 0.85: return 2.0
    return 1.0

def score_bias_gap(v):     # higher = better detection
    if v >= 0.05:  return 5.0
    if v >= 0.03:  return 4.0
    if v >= 0.010: return 3.0
    if v >= 0.005: return 2.0
    return 1.0

def score_disparity(v):    # model-level fairness surfaced by both methods
    if v >= 0.05: return 5.0
    if v >= 0.03: return 4.0
    if v >= 0.01: return 3.0
    return 2.0


def checkpoint(data, filename):
    with open(filename, 'w') as f:
        json.dump(data, f, indent=2)
    print(f"  [CHECKPOINT] {filename} saved.")


# -----------------------------------------------------------------------------
# STEP 1 — DATA LOADING AND PREPROCESSING
# -----------------------------------------------------------------------------
print("\n" + "=" * 65)
print("STEP 1: DATA LOADING AND PREPROCESSING")
print("=" * 65)

df = pd.read_csv('/content/application_train.csv')
print(f"Raw data shape: {df.shape}")

y = df['TARGET']
X = df.drop(['TARGET', 'SK_ID_CURR'], axis=1)

# Drop only 100%-missing columns (not the common 40%-threshold)
missing_frac = X.isnull().sum() / len(X)
cols_to_drop = missing_frac[missing_frac == 1.0].index.tolist()
X = X.drop(cols_to_drop, axis=1)
print(f"After dropping 100%-missing cols: {X.shape}")

# Label-encode all categorical columns
for col in X.select_dtypes(include='object').columns:
    le = LabelEncoder()
    X[col] = le.fit_transform(X[col].astype(str))

print(f"After encoding: {X.shape}")
print(f"Class balance — 0: {(y==0).sum()}  1: {(y==1).sum()}  "
      f"default rate: {y.mean()*100:.1f}%")

# Stratified split: 64% train / 16% val / 20% test
X_temp, X_test, y_temp, y_test = train_test_split(
    X, y, test_size=0.20, random_state=RANDOM_STATE, stratify=y)
X_train, X_val, y_train, y_val = train_test_split(
    X_temp, y_temp, test_size=0.20, random_state=RANDOM_STATE, stratify=y_temp)

print(f"\nSplit sizes — Train: {X_train.shape}  "
      f"Val: {X_val.shape}  Test: {X_test.shape}")

# Compute fill values from TRAINING SET ONLY — used everywhere
train_medians  = X_train.median(numeric_only=True)
X_train_filled = X_train.fillna(train_medians)
X_val_filled   = X_val.fillna(train_medians)
X_test_filled  = X_test.fillna(train_medians)

feature_names = list(X_train.columns)
print(f"Features: {len(feature_names)}")

# Save splits
X_train.to_csv('X_train.csv', index=False)
X_val.to_csv('X_val.csv', index=False)
X_test.to_csv('X_test.csv', index=False)
y_train.to_csv('y_train.csv', index=False)
y_val.to_csv('y_val.csv', index=False)
y_test.to_csv('y_test.csv', index=False)
print("Train/val/test CSVs saved.")


# -----------------------------------------------------------------------------
# STEP 2 — MODEL TRAINING AND SELECTION
# -----------------------------------------------------------------------------
print("\n" + "=" * 65)
print("STEP 2: MODEL TRAINING")
print("=" * 65)

# Imputer for LR and RF (they cannot handle NaN)
imputer     = SimpleImputer(strategy='median')
Xtr_imp     = imputer.fit_transform(X_train)
Xva_imp     = imputer.transform(X_val)
Xte_imp     = imputer.transform(X_test)

# -- Logistic Regression -------------------------------------------------------
print("\nTraining Logistic Regression...")
lr = LogisticRegression(max_iter=1000, class_weight='balanced',
                         random_state=RANDOM_STATE, solver='lbfgs')
lr.fit(Xtr_imp, y_train)
lr_prob  = lr.predict_proba(Xte_imp)[:, 1]
lr_pred  = (lr_prob >= 0.5).astype(int)
lr_auc   = roc_auc_score(y_test, lr_prob)
lr_prec  = precision_score(y_test, lr_pred, zero_division=0)
lr_rec   = recall_score(y_test, lr_pred, zero_division=0)
lr_f1    = f1_score(y_test, lr_pred, zero_division=0)
print(f"  AUC={lr_auc:.3f}  Prec={lr_prec:.3f}  "
      f"Rec={lr_rec:.3f}  F1={lr_f1:.3f}")

# -- Random Forest -------------------------------------------------------------
print("\nTraining Random Forest...")
rf = RandomForestClassifier(
    n_estimators=100, max_depth=10,
    class_weight='balanced',
    random_state=RANDOM_STATE, n_jobs=-1)
rf.fit(Xtr_imp, y_train)
rf_prob  = rf.predict_proba(Xte_imp)[:, 1]
rf_pred  = (rf_prob >= 0.5).astype(int)
rf_auc   = roc_auc_score(y_test, rf_prob)
rf_prec  = precision_score(y_test, rf_pred, zero_division=0)
rf_rec   = recall_score(y_test, rf_pred, zero_division=0)
rf_f1    = f1_score(y_test, rf_pred, zero_division=0)
print(f"  AUC={rf_auc:.3f}  Prec={rf_prec:.3f}  "
      f"Rec={rf_rec:.3f}  F1={rf_f1:.3f}")

# -- LightGBM (primary model) --------------------------------------------------
print("\nTraining LightGBM...")
scale_pw = (y_train == 0).sum() / (y_train == 1).sum()

lgb_train_ds = lgb.Dataset(X_train, label=y_train)
lgb_val_ds   = lgb.Dataset(X_val,   label=y_val, reference=lgb_train_ds)

lgb_params = {
    'objective':        'binary',
    'metric':           'auc',
    'boosting_type':    'gbdt',
    'num_leaves':       31,
    'learning_rate':    0.05,
    'feature_fraction': 0.9,
    'bagging_fraction': 0.8,
    'bagging_freq':     5,
    'scale_pos_weight': scale_pw,
    'verbose':          -1,
    'random_state':     RANDOM_STATE,
}

model = lgb.train(
    lgb_params, lgb_train_ds,
    num_boost_round=1000,
    valid_sets=[lgb_val_ds],
    callbacks=[lgb.early_stopping(50), lgb.log_evaluation(100)]
)

lgb_prob  = model.predict(X_test)
lgb_pred  = (lgb_prob >= 0.5).astype(int)
lgb_auc   = roc_auc_score(y_test, lgb_prob)
lgb_prec  = precision_score(y_test, lgb_pred, zero_division=0)
lgb_rec   = recall_score(y_test, lgb_pred, zero_division=0)
lgb_f1    = f1_score(y_test, lgb_pred, zero_division=0)
print(f"  AUC={lgb_auc:.3f}  Prec={lgb_prec:.3f}  "
      f"Rec={lgb_rec:.3f}  F1={lgb_f1:.3f}")

print(f"\n{'Model':<25} {'AUC':>6} {'Prec':>7} {'Rec':>7} {'F1':>7}")
print("-" * 55)
print(f"{'Logistic Regression':<25} {lr_auc:>6.3f} {lr_prec:>7.3f} "
      f"{lr_rec:>7.3f} {lr_f1:>7.3f}")
print(f"{'Random Forest':<25} {rf_auc:>6.3f} {rf_prec:>7.3f} "
      f"{rf_rec:>7.3f} {rf_f1:>7.3f}")
print(f"{'LightGBM ?':<25} {lgb_auc:>6.3f} {lgb_prec:>7.3f} "
      f"{lgb_rec:>7.3f} {lgb_f1:>7.3f}")
print("? Selected for XAI analysis (highest AUC, native TreeSHAP)")

model.save_model('lgbm_model.txt')

model_perf = {
    'logistic_regression': {'auc': lr_auc, 'precision': lr_prec,
                             'recall': lr_rec, 'f1': lr_f1},
    'random_forest':        {'auc': rf_auc, 'precision': rf_prec,
                             'recall': rf_rec, 'f1': rf_f1},
    'lightgbm':             {'auc': lgb_auc, 'precision': lgb_prec,
                             'recall': lgb_rec, 'f1': lgb_f1},
}
checkpoint(model_perf, 'model_performance.json')


# -----------------------------------------------------------------------------
# STEP 3 — PREPARE SHARED SAMPLES
# -----------------------------------------------------------------------------
print("\n" + "=" * 65)
print("STEP 3: PREPARING SHARED SAMPLES  (n=1000 for both SHAP and LIME)")
print("=" * 65)

# Both methods use the same 1000-instance sample drawn identically
# raw    = NaN present  ? for TreeSHAP (handles NaN natively)
# filled = NaN replaced ? for LIME (requires complete rows)
X_sample_raw    = X_test.sample(
    SAMPLE_N, random_state=RANDOM_STATE).reset_index(drop=True)
X_sample_filled = X_test_filled.sample(
    SAMPLE_N, random_state=RANDOM_STATE).reset_index(drop=True)

# Consistency subset — same rows, same order
X_consist_raw    = X_sample_raw.iloc[:N_CONSIST].copy()
X_consist_filled = X_sample_filled.iloc[:N_CONSIST].copy()

print(f"Main sample — raw NaN: {X_sample_raw.isnull().sum().sum()}")
print(f"Main sample — filled NaN: {X_sample_filled.isnull().sum().sum()}")
assert GENDER_COL in feature_names, f"{GENDER_COL} not found in features!"


# -----------------------------------------------------------------------------
# STEP 4 — SHAP ANALYSIS
# -----------------------------------------------------------------------------
print("\n" + "=" * 65)
print("STEP 4: SHAP ANALYSIS  (n=1000, TreeSHAP)")
print("=" * 65)

explainer   = shap.TreeExplainer(model)
shap_values = explainer.shap_values(X_sample_raw)
print(f"SHAP values shape: {shap_values.shape}")

# -- C1 Attribution Coverage ---------------------------------------------------
coverages = []
for i in range(len(shap_values)):
    abs_sv = np.abs(shap_values[i])
    total  = abs_sv.sum()
    if total == 0:
        continue
    coverages.append(np.sort(abs_sv)[::-1][:K].sum() / total)

shap_c1_mean = float(np.mean(coverages))
shap_c1_std  = float(np.std(coverages))
print(f"\nC1 Coverage:       {shap_c1_mean:.4f} ± {shap_c1_std:.4f}")

# -- C2 Consistency ------------------------------------------------------------
print("\nC2 Consistency (10 runs × 100 instances)...")
shap_runs = []
for run in range(N_RUNS):
    sv = explainer.shap_values(X_consist_raw)
    shap_runs.append([
        set(np.argsort(np.abs(sv[i]))[::-1][:K])
        for i in range(N_CONSIST)
    ])
    print(f"  Run {run+1}/{N_RUNS}")

shap_j_scores = [
    jaccard(shap_runs[r1][s], shap_runs[r2][s])
    for r1, r2 in combinations(range(N_RUNS), 2)
    for s in range(N_CONSIST)
]
shap_c2_mean = float(np.mean(shap_j_scores))
shap_c2_std  = float(np.std(shap_j_scores))
print(f"C2 Consistency:    {shap_c2_mean:.4f} ± {shap_c2_std:.4f}")

# -- C3a Structural Complexity -------------------------------------------------
shap_complexities = []
for i in range(len(shap_values)):
    top_names = [feature_names[idx]
                 for idx in np.argsort(np.abs(shap_values[i]))[::-1][:K]]
    shap_complexities.append(structural_complexity(top_names))

shap_c3a_mean = float(np.mean(shap_complexities))
shap_c3a_std  = float(np.std(shap_complexities))
print(f"C3a Complexity:    {shap_c3a_mean:.4f} ± {shap_c3a_std:.4f}")

# -- C4 Bias Detection (signed, unconditional) ---------------------------------
gender_idx   = feature_names.index(GENDER_COL)
gender_shap  = shap_values[:, gender_idx]          # signed, all instances
gender_vals  = X_sample_raw[GENDER_COL].values
valid_mask   = ~np.isnan(gender_vals.astype(float))
unique_g     = np.unique(gender_vals[valid_mask])

shap_group_means = {
    int(g): float(gender_shap[gender_vals == g].mean())
    for g in unique_g
}
sk = sorted(shap_group_means.keys())
shap_c4_gap = float(abs(shap_group_means[sk[0]] - shap_group_means[sk[1]]))
print(f"C4 Bias gap:       {shap_c4_gap:.4f}  groups: {shap_group_means}")

# -- C5 Fairness Transparency --------------------------------------------------
# LightGBM handles NaN natively; use raw sample for model predictions
preds_sample  = model.predict(X_sample_raw)
gender_c5     = X_sample_raw[GENDER_COL].values
shap_approval = {
    int(g): float((preds_sample[gender_c5 == g] < 0.5).mean())
    for g in unique_g
}
sk5          = sorted(shap_approval.keys())
shap_c5_disp = float(abs(shap_approval[sk5[0]] - shap_approval[sk5[1]]))
print(f"C5 Disparity:      {shap_c5_disp:.4f}  rates: {shap_approval}")

shap_results = {
    'c1_coverage_mean':    shap_c1_mean,
    'c1_coverage_std':     shap_c1_std,
    'c2_jaccard_mean':     shap_c2_mean,
    'c2_jaccard_std':      shap_c2_std,
    'c3a_complexity_mean': shap_c3a_mean,
    'c3a_complexity_std':  shap_c3a_std,
    'c4_bias_gap':         shap_c4_gap,
    'c4_group_means':      {str(k): v for k, v in shap_group_means.items()},
    'c5_disparity':        shap_c5_disp,
    'c5_approval_rates':   {str(k): v for k, v in shap_approval.items()},
}
checkpoint(shap_results, 'shap_results.json')


# -----------------------------------------------------------------------------
# STEP 5 — LIME ANALYSIS  (n=1000, all corrections applied)
# -----------------------------------------------------------------------------
print("\n" + "=" * 65)
print("STEP 5: LIME ANALYSIS  (n=1000 — this is the slow section ~2-4h)")
print("=" * 65)

lime_explainer = lime.lime_tabular.LimeTabularExplainer(
    training_data   = X_train_filled.values,
    feature_names   = feature_names,
    class_names     = ['Repaid', 'Default'],
    mode            = 'classification',
    discretize_continuous = True,
    random_state    = RANDOM_STATE
)

def predict_fn(x):
    df    = pd.DataFrame(x, columns=feature_names)
    preds = model.predict(df)
    return np.column_stack([1 - preds, preds])

# -- Generate 1000 explanations ------------------------------------------------
print(f"\nGenerating {SAMPLE_N} LIME explanations...")
lime_explanations = []
errors = 0
for i in range(SAMPLE_N):
    if i % 100 == 0:
        print(f"  {i}/{SAMPLE_N}  errors={errors}")
    try:
        exp = lime_explainer.explain_instance(
            data_row    = X_sample_filled.iloc[i].values,
            predict_fn  = predict_fn,
            num_features= K,
            num_samples = LIME_SAMPLES
        )
        lime_explanations.append({
            'features':  exp.as_list(),
            'r_squared': float(exp.score),
        })
    except Exception as e:
        errors += 1
        lime_explanations.append(None)

lime_explanations = [e for e in lime_explanations if e is not None]
n_lime = len(lime_explanations)
print(f"Successful: {n_lime}/{SAMPLE_N}  errors: {errors}")

# -- C1 Local Fidelity R² -----------------------------------------------------
r2_vals      = [e['r_squared'] for e in lime_explanations]
lime_c1_mean = float(np.mean(r2_vals))
lime_c1_std  = float(np.std(r2_vals))
print(f"\nC1 Fidelity R²:    {lime_c1_mean:.4f} ± {lime_c1_std:.4f}")

# -- C2 Stochastic Consistency -------------------------------------------------
print("\nC2 Consistency (10 stochastic runs × 100 instances)...")
lime_runs = []
for run in range(N_RUNS):
    print(f"  Stochastic run {run+1}/{N_RUNS}...")
    run_feats = []
    for i in range(N_CONSIST):
        try:
            exp = lime_explainer.explain_instance(
                data_row    = X_consist_filled.iloc[i].values,
                predict_fn  = predict_fn,
                num_features= K,
                num_samples = LIME_SAMPLES
            )
            run_feats.append(set(f[0] for f in exp.as_list()))
        except:
            run_feats.append(set())
    lime_runs.append(run_feats)

lime_j_scores = [
    jaccard(lime_runs[r1][s], lime_runs[r2][s])
    for r1, r2 in combinations(range(N_RUNS), 2)
    for s in range(N_CONSIST)
    if lime_runs[r1][s] or lime_runs[r2][s]
]
lime_c2_mean = float(np.mean(lime_j_scores))
lime_c2_std  = float(np.std(lime_j_scores))
print(f"C2 Consistency:    {lime_c2_mean:.4f} ± {lime_c2_std:.4f}")

# -- C3a Structural Complexity -------------------------------------------------
lime_complexities = [
    structural_complexity([str(f[0]) for f in exp['features']])
    for exp in lime_explanations
]
lime_c3a_mean = float(np.mean(lime_complexities))
lime_c3a_std  = float(np.std(lime_complexities))
print(f"C3a Complexity:    {lime_c3a_mean:.4f} ± {lime_c3a_std:.4f}")

# -- C4 Bias Detection — CORRECTED --------------------------------------------
# SIGNED weights; append 0.0 when gender is absent from top-10
gender_vals_lime   = X_sample_filled.iloc[:n_lime][GENDER_COL].values
gender_weights_sgn = {}
gender_appearances = 0

for i, exp in enumerate(lime_explanations):
    g = int(gender_vals_lime[i])
    if g not in gender_weights_sgn:
        gender_weights_sgn[g] = []
    found = False
    for feat_name, weight in exp['features']:
        if GENDER_COL.lower() in str(feat_name).lower():
            gender_weights_sgn[g].append(float(weight))  # SIGNED
            gender_appearances += 1
            found = True
            break
    if not found:
        gender_weights_sgn[g].append(0.0)                # ZERO-IMPUTE

lime_group_means        = {g: float(np.mean(w))
                            for g, w in gender_weights_sgn.items()}
lk                      = sorted(lime_group_means.keys())
lime_c4_gap             = float(abs(lime_group_means[lk[0]] -
                                    lime_group_means[lk[1]]))
lime_gender_appear_rate = float(gender_appearances / n_lime)

print(f"C4 Bias gap:       {lime_c4_gap:.6f}  groups: {lime_group_means}")
print(f"   Gender appear:  {gender_appearances}/{n_lime} "
      f"({lime_gender_appear_rate*100:.1f}%)")

# -- C5 Fairness Transparency --------------------------------------------------
# C5 is a model-level property; identical for both methods by definition
lime_approval = shap_approval
lime_c5_disp  = shap_c5_disp
print(f"C5 Disparity:      {lime_c5_disp:.4f}  (model-level, same as SHAP)")

lime_results = {
    'c1_fidelity_mean':      lime_c1_mean,
    'c1_fidelity_std':       lime_c1_std,
    'c2_jaccard_mean':       lime_c2_mean,
    'c2_jaccard_std':        lime_c2_std,
    'c3a_complexity_mean':   lime_c3a_mean,
    'c3a_complexity_std':    lime_c3a_std,
    'c4_bias_gap':           lime_c4_gap,
    'c4_group_means':        {str(k): v for k, v in lime_group_means.items()},
    'c4_gender_appear_rate': lime_gender_appear_rate,
    'c4_gender_appearances': gender_appearances,
    'c5_disparity':          lime_c5_disp,
    'c5_approval_rates':     {str(k): v for k, v in lime_approval.items()},
    'n_explanations':        n_lime,
}
checkpoint(lime_results, 'lime_results.json')


# -----------------------------------------------------------------------------
# STEP 6 — KERNEL WIDTH SWEEP  (Fig 1)

print("\n" + "=" * 65)
